Reading and Writing Numbers — Grade 1 Math

1. Technical Overview
This document specifies the technical architecture, component design, data models, and implementation standards for the Reading and Writing Numbers interactive lesson module (Lesson 1.2) for Intellia SG's Grade 1 Math program.
The module is a single-file React application rendered as a JSX artifact/component, designed to be embedded into the Intellia SG WordPress platform. It follows the visual theme established at https://number-counting-gamma.vercel.app/.

5. Randomisation Engine
5.1 Algorithm
javascriptfunction generateQuestionSet(bank, count = 100) {
  // Fisher-Yates shuffle for true randomisation
  const shuffled = [...bank];
  for (let i = shuffled.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
  }
  return shuffled.slice(0, count);
}

function generateMCQOptions(correct, range, count = 4) {
  // Ensure distractors are plausible (within ±5 of correct)
  const distractors = new Set();
  while (distractors.size < count - 1) {
    const d = correct + (Math.random() > 0.5 ? 1 : -1) * 
              Math.ceil(Math.random() * 5);
    if (d >= range.min && d <= range.max && d !== correct) {
      distractors.add(d);
    }
  }
  return shuffle([correct, ...distractors]);
}

8. Gamification Implementation
8.1 XP System
javascriptconst XP_VALUES = {
  correctFirstTry: 10,
  correctSecondTry: 7,
  correctWithHint: 5,
  streakBonus: (streak) => streak >= 5 ? 5 : 0,
};

function calculateXP(attempt, hintsUsed, streak) {
  const base = attempt === 1 ? 10 : hintsUsed === 0 ? 7 : 5;
  return base + XP_VALUES.streakBonus(streak);
}

9. Audio Implementation
9.1 Web Speech API Integration
javascriptfunction speak(text) {
  if (!window.speechSynthesis) return;
  const utterance = new SpeechSynthesisUtterance(text);
  utterance.rate = 0.85;     // Slightly slower for children
  utterance.pitch = 1.1;     // Slightly higher pitch
  utterance.lang = 'en-SG';  // Singapore English
  window.speechSynthesis.cancel();
  window.speechSynthesis.speak(utterance);
}

// Usage
speak('The number forty-three');
speak('That is correct! Well done!');

javascript// Use AudioContext for simple tones
const playTone = (frequency, duration = 200) => {
  const ctx = new AudioContext();
  const osc = ctx.createOscillator();
  osc.frequency.value = frequency;
  osc.connect(ctx.destination);
  osc.start();
  setTimeout(() => osc.stop(), duration);
};

const sounds = {
  correct: () => playTone(880),   // High A note
  wrong:   () => playTone(220),   // Low A note
  badge:   () => [523, 659, 784].forEach((f, i) => setTimeout(() => playTone(f), i * 150)),
};

10. Persistence & Storage
javascriptconst STORAGE_KEY = 'intellia_rw_numbers_v1';

const saveProgress = (state) => {
  localStorage.setItem(STORAGE_KEY, JSON.stringify({
    phase: state.phase,
    badges: state.badges,
    challengeScore: state.challengeScore,
    challengeXP: state.challengeXP,
    timestamp: Date.now(),
  }));
};

const loadProgress = () => {
  try {
    const data = JSON.parse(localStorage.getItem(STORAGE_KEY));
    if (data && (Date.now() - data.timestamp) < 24 * 60 * 60 * 1000) {
      return data; // Resume if within 24 hours
    }
  } catch (e) { /* fresh start */ }
  return null;
};
